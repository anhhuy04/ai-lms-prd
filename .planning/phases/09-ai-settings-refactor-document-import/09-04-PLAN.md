```dart
class TeacherFileDataSource {
  final SupabaseClient _supabase;
  TeacherFileDataSource(this._supabase);

  /// Step 1+2+3+4 of D-05: upload binary, save metadata, link to teacher, enqueue
  /// Returns TeacherFileModel with processingStatus='queued'
  Future<TeacherFileModel> uploadAndRegisterFile({
    required Uint8List bytes,
    required String filename,
    required String mimeType,
    required String teacherId,
  }) async {
    // Step 1: Upload to Storage
    final storagePath = 'teachers/$teacherId/$filename';
    await _supabase.storage
        .from('teacher-documents')
        .uploadBinary(storagePath, bytes,
            fileOptions: FileOptions(contentType: mimeType, upsert: true));

    final url = _supabase.storage
        .from('teacher-documents')
        .getPublicUrl(storagePath);  // signed URL for private bucket if needed

    // Step 2: INSERT into files table
    final fileRow = await _supabase
        .from('files')
        .insert({
          'storage_path': storagePath,
          'url': url,
          'filename': filename,
          'mime_type': mimeType,
          'size_bytes': bytes.length,
          'uploaded_by': teacherId,
        })
        .select()
        .single();

    final fileId = fileRow['id'] as String;

    // Step 3: INSERT file_links (target_type='teacher', target_id=teacherId)
    await _supabase.from('file_links').insert({
      'file_id': fileId,
      'target_type': 'teacher',
      'target_id': teacherId,
    });

    // Step 4: INSERT ai_queue (fire-and-forget — UI does NOT wait)
    await _supabase.from('ai_queue').insert({
      'request_type': 'vectorize_document',
      'request_payload': {'file_id': fileId, 'action': 'extract_or_vectorize'},
      'status': 'pending',
      'attempts': 0,
    });

    // Step 5: Return immediately — UI can show chip
    return TeacherFileModel.fromJson({
      ...fileRow,
      'processing_status': 'queued',
    });
  }

  /// Query teacher's files via file_links join
  Future<List<TeacherFileModel>> getTeacherFiles(String teacherId) async {
    final rows = await _supabase
        .from('file_links')
        .select('files(*)')
        .eq('target_type', 'teacher')
        .eq('target_id', teacherId)
        .order('created_at', ascending: false);

    return rows
        .map((r) => TeacherFileModel.fromJson(r['files'] as Map<String, dynamic>))
        .toList();
  }
}
```

class TeacherFileRepositoryImpl implements ITeacherFileRepository {
  final TeacherFileDataSource _dataSource;
  final String _teacherId;

  TeacherFileRepositoryImpl(this._dataSource, this._teacherId);

  @override
  Future<TeacherFileModel> uploadFile({...}) =>
      _dataSource.uploadAndRegisterFile(

  @override
  Future<List<TeacherFileModel>> getTeacherFiles() =>
      _dataSource.getTeacherFiles(_teacherId);
}

@riverpod
class TeacherFileNotifier extends _$TeacherFileNotifier {
  @override
  Future<List<TeacherFileModel>> build() async {
    return ref.read(teacherFileRepositoryProvider).getTeacherFiles();
  }

  Future<void> uploadFile(Uint8List bytes, String filename, String mimeType) async {
    // Optimistic UI: add loading indicator
    state = const AsyncLoading();
    state = await AsyncValue.guard(() async {
      final repo = ref.read(teacherFileRepositoryProvider);
      final newFile = await repo.uploadFile(bytes: bytes, filename: filename, mimeType: mimeType);
      final current = state.valueOrNull ?? [];
      return [newFile, ...current];
    });
  }
}

@riverpod
ITeacherFileRepository teacherFileRepository(TeacherFileRepositoryRef ref) {
  final supabase = ref.read(supabaseClientProvider);
  final authState = ref.read(authStateProvider);
  final teacherId = authState.session?.user.id ?? '';
  return TeacherFileRepositoryImpl(TeacherFileDataSource(supabase), teacherId);
}
