```dart
import 'package:freezed_annotation/freezed_annotation.dart';

part 'question_dto.freezed.dart';

/// Unified DTO for AI-generated/extracted questions.
/// Both Extraction pipeline and Generation pipeline produce this schema (D-24, D-25).
/// Pipeline agnosticism: Flutter frontend does not know which pipeline ran.
@freezed
class QuestionDTO with _$QuestionDTO {
  const factory QuestionDTO({
    @Default('multiple_choice') String type,  // multiple_choice | true_false | short_answer
    required Map<String, dynamic> content,    // {"text": "Question text"}
    @Default([]) List<ChoiceDTO> choices,     // Empty for non-MC questions
    required Map<String, dynamic> answer,     // {"correct_index": 0} or {"correct_text": "..."}
    @Default('medium') String difficulty,
    @Default([]) List<String> tags,
    @Default(1) int defaultPoints,
  }) = _QuestionDTO;

  factory QuestionDTO.fromJson(Map<String, dynamic> json) =>
      _$QuestionDTOFromJson(json);
}

@freezed
class ChoiceDTO with _$ChoiceDTO {
  const factory ChoiceDTO({
    required int id,
    required String text,
    @Default(false) bool isCorrect,
  }) = _ChoiceDTO;

  factory ChoiceDTO.fromJson(Map<String, dynamic> json) =>
      _$ChoiceDTOFromJson(json);
}
```

```dart
Future<void> _saveToBank() async {
  if (_isSaving) return;
  setState(() => _isSaving = true);
  try {
    await supabase.rpc('save_questions_to_assignment', params: {
      'p_questions': questions.map((q) => q.toDbInsert()).toList(),
      'p_assignment_id': null,  // bank only
    });
    if (mounted) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Đã lưu ${questions.length} câu vào Ngân hàng')));
      widget.onComplete();
      Navigator.of(context).pop();
    }
  } catch (e) {
    if (mounted) ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Lỗi lưu: $e'), backgroundColor: DesignColors.error));
  } finally {
    if (mounted) setState(() => _isSaving = false);
  }
}

Future<void> _saveAndAddToAssignment() async {
  if (_isSaving || widget.assignmentId == null) return;
  setState(() => _isSaving = true);
  try {
    await supabase.rpc('save_questions_to_assignment', params: {
      'p_questions': questions.map((q) => q.toDbInsert()).toList(),
      'p_assignment_id': widget.assignmentId,  // atomic: save + link
    });
    if (mounted) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Đã lưu và thêm ${questions.length} câu vào Đề thi')));
      widget.onComplete();
      Navigator.of(context).pop();
    }
  } catch (e) {
    if (mounted) ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Lỗi lưu: $e'), backgroundColor: DesignColors.error));
  } finally {
    if (mounted) setState(() => _isSaving = false);
  }
}
```

Add a polling mechanism: when user taps "Tạo câu hỏi từ tài liệu" (document source), instead of calling AI inline, wait for the `ai_queue` row to transition to `status='done'`, then read `result.questions`:

```dart
Future<void> _pollForDocumentResults(String queueId) async {
  // Poll every 3s, max 60s
  for (int attempt = 0; attempt < 20; attempt++) {
    await Future.delayed(const Duration(seconds: 3));
    final row = await supabase.from('ai_queue')
        .select('status, result').eq('id', queueId).single();
    if (row['status'] == 'done') {
      final rawQuestions = row['result']['questions'] as List;
      final questions = rawQuestions
          .map((q) => QuestionDTO.fromJson(q as Map<String, dynamic>))
          .toList();
      if (mounted) {
        showStagingArea(context,
          questions: questions,
          assignmentId: widget.assignmentId,  // from screen param
          onComplete: () => setState(() => _generatedQuestions = questions));
      }
      return;
    }
    if (row['status'] == 'error') throw Exception('Document processing failed');
  }
  throw Exception('Document processing timed out');
}
```
