```dart
import 'package:freezed_annotation/freezed_annotation.dart';

part 'recommendation.freezed.dart';

/// Recommendation entity - single sink for all recommendations (rule-based + AI)
/// Phase 5: Rule-based recommendations
/// Phase 6: AI-generated recommendations via ai_queue
@freezed
class Recommendation with _$Recommendation {
  const factory Recommendation({
    required String id,
    String? teacherId,
    String? classId,
    String? studentId,
    @Default('individual') String type,
      // 'individual' | 'small_group' | 'class'
    @Default(3) int priority,
      // 1-5 (1 = most urgent, 2 = high, 3 = medium, 4 = low, 5 = info)
    required String title,
    String? description,
    Map<String, dynamic>? resources,
      // JSONB: {"exercises": ["uuid"], "videos": ["url"], "documents": ["url"]}
    @Default(false) bool dismissed,
    DateTime? createdAt,
  }) = _Recommendation;

  factory Recommendation.fromJson(Map<String, dynamic> json) =>
      _$RecommendationFromJson(json);
}

/// Peer comparison data from RPC (anonymous - no raw scores sent to client)
@freezed
class PeerComparison with _$PeerComparison {
  const factory PeerComparison({
    @Default(0.0) @JsonKey(name: 'class_average') double classAverage,
    @Default(0.0) double percentile,
    @Default(0) int rank,
    @Default(0) @JsonKey(name: 'total_students') int totalStudents,
  }) = _PeerComparison;

  factory PeerComparison.fromJson(Map<String, dynamic> json) =>
      _$PeerComparisonFromJson(json);
}

/// Recommendation type enum
enum RecommendationType {
  individual,
  smallGroup,
  classType,
}

/// Recommendation priority levels
enum RecommendationPriority {
  critical(1),
  high(2),
  medium(3),
  low(4),
  info(5);

  final int value;
  const RecommendationPriority(this.value);
}

/// Extension helpers on Recommendation
extension RecommendationHelpers on Recommendation {
  /// Returns exercise UUIDs from resources JSONB
  List<String> get exercises =>
    (resources?['exercises'] as List<dynamic>?)?.cast<String>() ?? [];

  /// Returns video URLs from resources JSONB
  List<String> get videos =>
    (resources?['videos'] as List<dynamic>?)?.cast<String>() ?? [];

  /// Returns document URLs from resources JSONB
  List<String> get documents =>
    (resources?['documents'] as List<dynamic>?)?.cast<String>() ?? [];

  /// Whether this is an urgent recommendation (priority 1-2)
  bool get isUrgent => priority <= 2;

  /// Whether this has learning resources attached
  bool get hasResources =>
    exercises.isNotEmpty || videos.isNotEmpty || documents.isNotEmpty;

  /// Recommendation type as enum
  RecommendationType get recommendationType {
    switch (type) {
      case 'small_group':
        return RecommendationType.smallGroup;
      case 'class':
        return RecommendationType.classType;
      default:
        return RecommendationType.individual;
    }
  }
}
```

```dart
import 'package:ai_mls/core/services/supabase_service.dart';
import 'package:ai_mls/core/utils/app_logger.dart';
import 'package:ai_mls/domain/entities/recommendation.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

/// DataSource for recommendation queries from Supabase ai_recommendations table.
/// Dumb UI pattern: frontend only SELECTs, generation logic lives in DB triggers/cron.
class RecommendationDatasource {
  SupabaseClient get _client => SupabaseService.client;

  /// Get recommendations for a user.
  /// When forTeacher=true, filters by teacherId; otherwise filters by studentId.
  /// Sorted by priority ASC (most urgent first), then createdAt DESC.
  Future<List<Recommendation>> getRecommendations({
    String? userId,
    required bool forTeacher,
    int? limit,
  }) async {
    try {
      var q = _client
          .from('ai_recommendations')
          .select()
          .eq('dismissed', false);

      if (forTeacher) {
        q = q.eq('teacher_id', userId);
      } else {
        q = q.eq('student_id', userId);
      }

      q = q.order('priority', ascending: true)
           .order('created_at', ascending: false);

      if (limit != null) {
        q = q.limit(limit);
      }

      final result = await q;
      return result.map((row) => Recommendation.fromJson(row)).toList();
    } catch (e, st) {
      AppLogger.error(
        '[RecommendationDatasource] getRecommendations error',
        error: e,
        stackTrace: st,
      );
      return [];
    }
  }

  /// Count of urgent recommendations (priority <= 2) for teacher badge.
  Future<int> getInterventionCount(String teacherId) async {
    try {
      final result = await _client
          .from('ai_recommendations')
          .select('id')
          .eq('teacher_id', teacherId)
          .eq('dismissed', false)
          .lte('priority', 2);
      return result.length;
    } catch (e, st) {
      AppLogger.error(
        '[RecommendationDatasource] getInterventionCount error',
        error: e,
        stackTrace: st,
      );
      return 0;
    }
  }

  /// Dismiss a recommendation (set dismissed = true).
  Future<void> dismissRecommendation(String recommendationId) async {
    try {
      await _client
          .from('ai_recommendations')
          .update({'dismissed': true})
          .eq('id', recommendationId);
    } catch (e, st) {
      AppLogger.error(
        '[RecommendationDatasource] dismissRecommendation error',
        error: e,
        stackTrace: st,
      );
    }
  }

  /// Get peer comparison data for a student in a class.
  /// Uses Supabase RPC (SECURITY DEFINER) to compute percentile + class average
  /// server-side. NO raw scores are ever sent to the client.
  /// Per REC-03: only anonymous aggregates (percentile, class_average) returned.
  Future<PeerComparison> getPeerComparison(
    String studentId,
    String classId,
  ) async {
    try {
      final result = await _client.rpc(
        'get_student_peer_comparison',
        params: {
          'p_student_id': studentId,
          'p_class_id': classId,
        },
      );

      if (result == null) {
        return const PeerComparison();
      }

      return PeerComparison.fromJson(result as Map<String, dynamic>);
    } catch (e, st) {
      AppLogger.error(
        '[RecommendationDatasource] getPeerComparison error',
        error: e,
        stackTrace: st,
      );
      return const PeerComparison();
    }
  }

  /// Get skill mastery data for class average comparison (for DualRadarChart).
  /// Returns map of objectiveId -> class average mastery_level (0.0-1.0).
  Future<Map<String, double>> getClassAverageSkillMastery(
    String classId,
  ) async {
    try {
      // Get all approved students in this class
      final studentsResult = await _client
          .from('class_members')
          .select('student_id')
          .eq('class_id', classId)
          .eq('status', 'approved');
      final studentIds = studentsResult
          .map((s) => s['student_id'] as String)
          .toList();

      if (studentIds.isEmpty) return {};

      // Get skill mastery for all class members
      final masteryResult = await _client
          .from('student_skill_mastery')
          .select('objective_id, mastery_level')
          .inFilter('student_id', studentIds);

      if (masteryResult.isEmpty) return {};

      // Group by objective_id and compute average
      final grouped = <String, List<double>>{};
      for (final row in masteryResult) {
        final oid = row['objective_id'] as String;
        final level = (row['mastery_level'] ?? 0.0) as num;
        grouped.putIfAbsent(oid, () => []).add(level.toDouble());
      }

      return grouped.map((oid, levels) =>
        MapEntry(oid, levels.reduce((a, b) => a + b) / levels.length)
      );
    } catch (e, st) {
      AppLogger.error(
        '[RecommendationDatasource] getClassAverageSkillMastery error',
        error: e,
        stackTrace: st,
      );
      return {};
    }
  }
}
```

Create `db/migration_05_recommendations.sql`:

```sql
-- ==============================================================================
-- MIGRATION: Phase 5 - Personalized Recommendations
-- Created: 2026-03-25
-- Purpose: RPC for peer comparison (REC-03) + RLS policies for ai_recommendations
-- ==============================================================================

-- ══════════════════════════════════════════════════════════════════════════════
-- 1. RPC: get_student_peer_comparison (SECURITY DEFINER)
--    Computes percentile and class average server-side.
--    Per REC-03: NO raw scores ever sent to the Flutter client.
-- ══════════════════════════════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION public.get_student_peer_comparison(
  p_student_id UUID,
  p_class_id UUID
) RETURNS JSONB
SECURITY DEFINER
SET search_path = public
LANGUAGE plpgsql
AS $$
DECLARE
  v_result JSONB;
  v_student_avg NUMERIC;
  v_class_avg NUMERIC;
  v_percentile NUMERIC;
  v_rank INTEGER;
  v_total INTEGER;
BEGIN
  -- Step 1: Get this student's average score for this class's assignments
  WITH class_distributions AS (
    SELECT id FROM public.assignment_distributions
    WHERE class_id = p_class_id
  )
  SELECT COALESCE(AVG(s.total_score), 0)
  INTO v_student_avg
  FROM public.submissions s
  WHERE s.student_id = p_student_id
    AND s.assignment_distribution_id IN (SELECT id FROM class_distributions)
    AND s.total_score IS NOT NULL;

  -- Step 2: Compute per-student averages for this class
  WITH class_distributions AS (
    SELECT id FROM public.assignment_distributions
    WHERE class_id = p_class_id
  ),
  student_avgs AS (
    SELECT
      s.student_id,
      AVG(s.total_score) AS avg_score
    FROM public.submissions s
    WHERE s.assignment_distribution_id IN (SELECT id FROM class_distributions)
      AND s.total_score IS NOT NULL
    GROUP BY s.student_id
  ),
  ranked AS (
    SELECT
      avg_score,
      ROW_NUMBER() OVER (ORDER BY avg_score DESC) AS rn,
      COUNT(*) OVER () AS total
    FROM student_avgs
  )
  SELECT
    (SELECT AVG(avg_score) FROM student_avgs),
    (SELECT ROUND(
      COUNT(*) FILTER (WHERE avg_score < v_student_avg) * 100.0 / NULLIF(MAX(r.total), 0), 1
    ) FROM ranked r),
    (SELECT COALESCE(MIN(rn), 1) FROM ranked r WHERE r.avg_score <= v_student_avg),
    (SELECT MAX(total) FROM ranked)
  INTO v_class_avg, v_percentile, v_rank, v_total
  FROM ranked LIMIT 1;

  RETURN jsonb_build_object(
    'class_average', ROUND(COALESCE(v_class_avg, 0)::numeric, 2),
    'percentile', ROUND(COALESCE(v_percentile, 0)::numeric, 1),
    'rank', COALESCE(v_rank, 0),
    'total_students', COALESCE(v_total, 0)
  );
END;
$$;

-- ══════════════════════════════════════════════════════════════════════════════
-- 2. RLS Policies for ai_recommendations table
--    Teachers can see their own recommendations.
--    Students can see their own recommendations.
-- ══════════════════════════════════════════════════════════════════════════════

-- Enable RLS on ai_recommendations
ALTER TABLE public.ai_recommendations ENABLE ROW LEVEL SECURITY;

-- Policy: Teachers can read recommendations where teacher_id matches auth.uid()
CREATE POLICY "teachers_read_own_recommendations" ON public.ai_recommendations
  FOR SELECT
  USING (teacher_id = (SELECT auth.uid()));

-- Policy: Teachers can update (dismiss) their own recommendations
CREATE POLICY "teachers_update_own_recommendations" ON public.ai_recommendations
  FOR UPDATE
  USING (teacher_id = (SELECT auth.uid()))
  WITH CHECK (teacher_id = (SELECT auth.uid()));

-- Policy: Students can read their own recommendations (student_id = auth.uid())
CREATE POLICY "students_read_own_recommendations" ON public.ai_recommendations
  FOR SELECT
  USING (student_id = (SELECT auth.uid()));

CREATE POLICY "students_update_own_recommendations" ON public.ai_recommendations
  FOR UPDATE
  USING (student_id = (SELECT auth.uid()))
  WITH CHECK (student_id = (SELECT auth.uid()));
```

```dart
/// Dismiss a recommendation (teacher or student, based on auth)
@riverpod
class DismissRecommendation extends _$DismissRecommendation {
  @override
  Future<void> build({required String recommendationId}) async {}

  Future<void> dismiss() async {
    try {
      final repo = ref.read(recommendationRepositoryProvider);
      await repo.dismissRecommendation(recommendationId);
      // Invalidate the relevant list providers to refresh UI
      ref.invalidate(teacherRecommendationsProvider);
      ref.invalidate(studentRecommendationsProvider);
      ref.invalidate(top3RecommendationsProvider);
      ref.invalidate(interventionCountProvider);
    } catch (e, st) {
      AppLogger.error('[DismissRecommendation] Error', error: e, stackTrace: st);
      rethrow;
    }
  }
}
```

```dart
        },
        'dismissed': false,
        'created_at': '2026-03-25T10:00:00Z',
      };

      final rec = Recommendation.fromJson(json);

      expect(rec.id, 'test-id');
      expect(rec.teacherId, 'teacher-uuid');
      expect(rec.classId, 'class-uuid');
      expect(rec.studentId, 'student-uuid');
      expect(rec.type, 'individual');

      expect(const Recommendation(id: '1', title: 'a', priority: 2).isUrgent, true);
      expect(const Recommendation(id: '1', title: 'a', priority: 3).isUrgent, false);
    });

    test('exercises returns exercise UUIDs from resources', () {
      final rec = Recommendation(
        id: '1',
        title: 'test',
        resources: {'exercises': ['uuid-1', 'uuid-2'], 'videos': [], 'documents': []},
      );
      expect(rec.exercises, ['uuid-1', 'uuid-2']);
      expect(rec.videos, isEmpty);
    });

    test('hasResources returns true when resources exist', () {
      expect(
        Recommendation(id: '1', title: 'a', resources: {'exercises': ['x'], 'videos': [], 'documents': []}).hasResources,
        true,
      );
      expect(
        const Recommendation(id: '1', title: 'a').hasResources,
        false,
      );
    });
  });
```
