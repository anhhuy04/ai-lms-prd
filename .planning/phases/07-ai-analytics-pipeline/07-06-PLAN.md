import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

interface AiQueueItem {
  id: string;
  submission_answer_id: string;
  request_type: "score" | "feedback" | "analysis";
  status: string;
  attempts: number;
  payload: Record<string, unknown> | null;
  result: Record<string, unknown> | null;
  created_at: string;
}

Deno.serve(async (req: Request) => {
  try {
    const supabase = createClient(
      Deno.env.get("SUPABASE_URL")!,
      Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!  // D-20: bypass RLS
    );

    // Parse optional body for retroactive processing
    let body: Record<string, unknown> = {};
    try {
      body = await req.json();
    } catch {
      // No body — process general queue
    }

    const submissionId = body.submission_id as string | undefined;
    const retroactive = body.retroactive as boolean | undefined;

    // Fetch pending items from ai_queue
    let query = supabase
      .from("ai_queue")
      .select("*")
      .eq("status", "pending")
      .order("created_at")
      .limit(10);

    // If retroactive for specific submission, filter by submission answers
    if (submissionId && retroactive) {
      const { data: answerIds } = await supabase
        .from("submission_answers")
        .select("id")
        .eq("session_id", submissionId);
      if (answerIds && answerIds.length > 0) {
        query = query.in(
          "submission_answer_id",
          answerIds.map((a: { id: string }) => a.id)
        );
      }
    }

    const { data: items, error: fetchError } = await query;
    if (fetchError) {
      console.error("Failed to fetch ai_queue:", fetchError);
      return new Response(
        JSON.stringify({ error: fetchError.message }),
        { status: 500 }
      );
    }

    let processed = 0;
    for (const item of (items as AiQueueItem[]) ?? []) {
      // Mark processing
      await supabase
        .from("ai_queue")
        .update({ status: "processing", attempts: item.attempts + 1, updated_at: new Date().toISOString() })
        .eq("id", item.id);

      try {
        if (item.request_type === "feedback") {
          await handleFeedback(supabase, item);
        } else if (item.request_type === "analysis") {
          await handleAnalysis(supabase, item);
        } else if (item.request_type === "score") {
          // STUB: Essay scoring deferred until Phase 3 re-enable
          console.log(`[STUB] Score request deferred: ${item.id}`);
        }

        // Mark completed
        await supabase
          .from("ai_queue")
          .update({ status: "completed", updated_at: new Date().toISOString() })
          .eq("id", item.id);
        processed++;
      } catch (error) {
        console.error(`[AI] Failed to process item ${item.id}:`, error);
        // Retry logic: max 3 attempts
        if (item.attempts + 1 >= 3) {
          await supabase
            .from("ai_queue")
            .update({ status: "failed", updated_at: new Date().toISOString() })
            .eq("id", item.id);
        } else {
          await supabase
            .from("ai_queue")
            .update({ status: "pending", updated_at: new Date().toISOString() })
            .eq("id", item.id);
        }
      }
    }

    return new Response(
      JSON.stringify({ processed, total: items?.length ?? 0 }),
      { headers: { "Content-Type": "application/json" } }
    );
  } catch (error) {
    console.error("[AI] Edge function error:", error);
    return new Response(
      JSON.stringify({ error: String(error) }),
      { status: 500 }
    );
  }
});

// FEEDBACK HANDLER (7-08): Explain MCQ correct/incorrect answers
// ============================================================
async function handleFeedback(
  supabase: ReturnType<typeof createClient>,
  item: AiQueueItem
) {
  // D-21: Self-JOIN to get question context
  const { data: context } = await supabase
    .from("submission_answers")
    .select(`
      id,
      answer,
      final_score,
      assignment_question_id,
      session_id,
      assignment_questions!inner (
        points,
        question_id,
        questions!inner (
          question_text,
          question_choices (
            id,
            choice_text,
            is_correct
          )
        )
      )
    `)
    .eq("id", item.submission_answer_id)
    .single();

  if (!context) {
    throw new Error(`Submission answer not found: ${item.submission_answer_id}`);
  }

  // D-18: Get teacher AI config from profiles.metadata.analytics
  const { data: session } = await supabase
    .from("work_sessions")
    .select("student_id, assignment_distribution_id")
    .eq("id", context.session_id)
    .single();

  const { data: distribution } = await supabase
    .from("assignment_distributions")
    .select("assignment_id, assignments!inner(teacher_id)")
    .eq("id", session?.assignment_distribution_id)
    .single();

  const teacherId = (distribution?.assignments as Record<string, unknown>)?.teacher_id as string;

  const { data: profile } = await supabase
    .from("profiles")
    .select("metadata")
    .eq("id", teacherId)
    .single();

  const metadata = profile?.metadata as Record<string, unknown> | null;
  const analyticsConfig = metadata?.analytics as Record<string, string> | null;
  const provider = analyticsConfig?.provider ?? "gemini";
  const model = analyticsConfig?.model ?? "gemini-2.0-flash";
  const apiKeys = metadata?.api_keys as Record<string, string> | null;
  const apiKey = apiKeys?.[provider];

  if (!apiKey) {
    console.warn(`[AI] No API key for provider "${provider}" — teacher ${teacherId}`);
    // Write fallback feedback
    await supabase
      .from("submission_answers")
      .update({
        ai_feedback: {
          status: "no_api_key",
          message: "Giao vien chua cau hinh API key cho AI phan tich.",
        },
      })
      .eq("id", item.submission_answer_id);
    return;
  }

  // Build prompt for MCQ feedback
  const aq = context.assignment_questions as Record<string, unknown>;
  const q = aq.questions as Record<string, unknown>;
  const choices = q.question_choices as Array<Record<string, unknown>>;
  const studentAnswer = context.answer as Record<string, unknown>;
  const selectedIds = (studentAnswer?.selected_choice_ids as string[]) ?? [];
  const correctChoices = choices.filter((c) => c.is_correct);
  const selectedChoices = choices.filter((c) => selectedIds.includes(c.id as string));
  const isCorrect = context.final_score === (aq.points as number);

  const prompt = `Analyze this student's MCQ answer and provide educational feedback in Vietnamese.

Question: ${q.question_text}
Choices: ${choices.map((c) => `- ${c.choice_text} ${c.is_correct ? "(correct)" : ""}`).join("\n")}
Student selected: ${selectedChoices.map((c) => c.choice_text).join(", ") || "Nothing"}
Result: ${isCorrect ? "Correct" : "Incorrect"}

Provide:
1. Brief explanation of why the correct answer is correct (2-3 sentences)
2. If student was wrong, explain the misconception (1-2 sentences)
3. A learning tip related to this concept (1 sentence)

Respond in Vietnamese. Be encouraging and educational.`;

  // Call AI API based on provider
  const feedback = await callAiApi(provider, model, apiKey, prompt);

  // D-19: Write feedback to submission_answers.ai_feedback
  await supabase
    .from("submission_answers")
    .update({
      ai_feedback: {
        status: "completed",
        provider: provider,
        model: model,
        content: feedback,
        is_correct: isCorrect,
        correct_answer: correctChoices.map((c) => c.choice_text).join(", "),
        generated_at: new Date().toISOString(),
      },
    })
    .eq("id", item.submission_answer_id);
}

// ANALYSIS HANDLER (7-07): Generate recommendations from skill gaps
// ============================================================
async function handleAnalysis(
  supabase: ReturnType<typeof createClient>,
  item: AiQueueItem
) {
  // D-14: Dual-channel — student recommendations + teacher interventions
  // Get student and their skill mastery data
  const { data: answer } = await supabase
    .from("submission_answers")
    .select("session_id")
    .eq("id", item.submission_answer_id)
    .single();

  if (!answer) return;

  const { data: session } = await supabase
    .from("work_sessions")
    .select("student_id, assignment_distribution_id")
    .eq("id", answer.session_id)
    .single();

  if (!session) return;

  const studentId = session.student_id as string;

  // Get low mastery skills (< 0.6 threshold per D-14)
  const { data: lowMastery } = await supabase
    .from("student_skill_mastery")
    .select("objective_id, mastery_level, attempts, learning_objectives!inner(name, subject_code)")
    .eq("student_id", studentId)
    .lt("mastery_level", 0.6);

  if (!lowMastery || lowMastery.length === 0) return;

  // Get class and teacher info from distribution
  const { data: dist } = await supabase
    .from("assignment_distributions")
    .select("class_id, assignments!inner(teacher_id)")
    .eq("id", session.assignment_distribution_id)
    .single();

  if (!dist) return;

  const teacherId = (dist.assignments as Record<string, unknown>)?.teacher_id as string;
  const classId = dist.class_id as string;

  // D-14 Channel 1: Student recommendation (learning resources)
  for (const skill of lowMastery) {
    const objective = skill.learning_objectives as Record<string, unknown>;
    const objectiveName = objective.name as string;
    const masteryPct = Math.round((skill.mastery_level as number) * 100);

    // Check if recommendation already exists for this student+objective (avoid duplicates)
    const { data: existing } = await supabase
      .from("ai_recommendations")
      .select("id")
      .eq("student_id", studentId)
      .eq("title", `Ren luyen: ${objectiveName}`)
      .eq("dismissed", false)
      .maybeSingle();

    if (existing) continue; // Skip if active recommendation exists

    await supabase.from("ai_recommendations").insert({
      teacher_id: teacherId,
      class_id: classId,
      student_id: studentId,
      type: "individual",
      priority: masteryPct < 30 ? 5 : masteryPct < 50 ? 4 : 3,
      title: `Ren luyen: ${objectiveName}`,
      description: `Hoc sinh dat ${masteryPct}% (${skill.attempts} lan thu). Can on tap them ve "${objectiveName}".`,
      resources: { objective_id: skill.objective_id, mastery_level: skill.mastery_level },
    });
  }

  // D-14 Channel 2: Teacher intervention (if multiple low-mastery skills)
  if (lowMastery.length >= 3) {
    const { data: existingIntervention } = await supabase
      .from("ai_recommendations")
      .select("id")
      .eq("teacher_id", teacherId)
      .eq("class_id", classId)
      .eq("student_id", studentId)
      .eq("type", "individual")
      .ilike("title", "%Can thiep%")
      .eq("dismissed", false)
      .maybeSingle();

    if (!existingIntervention) {
      const skillNames = lowMastery
        .map((s) => (s.learning_objectives as Record<string, unknown>).name)
        .slice(0, 5)
        .join(", ");

      await supabase.from("ai_recommendations").insert({
        teacher_id: teacherId,
        class_id: classId,
        student_id: studentId,
        type: "individual",
        priority: 5,
        title: `Can thiep: Hoc sinh yeu nhieu ki nang`,
        description: `Hoc sinh co ${lowMastery.length} ki nang duoi 60%: ${skillNames}. Can ho tro ca nhan.`,
        resources: {
          low_mastery_count: lowMastery.length,
          skills: lowMastery.map((s) => ({
            objective_id: s.objective_id,
            mastery: s.mastery_level,
          })),
        },
      });
    }
  }
}

// AI API CALLER — Supports Gemini, Groq, Ollama
// ============================================================
async function callAiApi(
  provider: string,
  model: string,
  apiKey: string,
  prompt: string
): Promise<string> {
  let url: string;
  let headers: Record<string, string>;
  let bodyObj: Record<string, unknown>;

  if (provider === "gemini") {
    url = `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${apiKey}`;
    headers = { "Content-Type": "application/json" };
    bodyObj = {
      contents: [{ parts: [{ text: prompt }] }],
      generationConfig: { maxOutputTokens: 500, temperature: 0.3 },
    };
  } else if (provider === "groq") {
    url = "https://api.groq.com/openai/v1/chat/completions";
    headers = {
      "Content-Type": "application/json",
      Authorization: `Bearer ${apiKey}`,
    };
    bodyObj = {
      model: model,
      messages: [{ role: "user", content: prompt }],
      max_tokens: 500,
      temperature: 0.3,
    };
  } else if (provider === "ollama") {
    // Ollama typically runs locally — apiKey is the base URL
    url = `${apiKey}/api/generate`;
    headers = { "Content-Type": "application/json" };
    bodyObj = { model: model, prompt: prompt, stream: false };
  } else {
    throw new Error(`Unsupported AI provider: ${provider}`);
  }

  const response = await fetch(url, {
    method: "POST",
    headers,
    body: JSON.stringify(bodyObj),
  });

  if (!response.ok) {
    const errorText = await response.text();
    throw new Error(`AI API error (${provider}): ${response.status} - ${errorText}`);
  }

  const data = await response.json();

  // Extract text based on provider response format
  if (provider === "gemini") {
    return data.candidates?.[0]?.content?.parts?.[0]?.text ?? "Khong co phan hoi.";
  } else if (provider === "groq") {
    return data.choices?.[0]?.message?.content ?? "Khong co phan hoi.";
  } else if (provider === "ollama") {
    return data.response ?? "Khong co phan hoi.";
  }
  return "Khong co phan hoi.";
}
