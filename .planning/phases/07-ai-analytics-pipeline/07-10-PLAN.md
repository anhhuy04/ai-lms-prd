**Custom JSON converter for unknown values (D-16):**

Add a `SubmissionStatusConverter` class that handles unknown status strings gracefully:

```dart
class SubmissionStatusConverter implements JsonConverter<SubmissionStatus, String?> {
  const SubmissionStatusConverter();

  @override
  SubmissionStatus fromJson(String? json) {
    if (json == null) return SubmissionStatus.unknown;
    switch (json) {
      case 'draft': return SubmissionStatus.draft;
      case 'submitted': return SubmissionStatus.submitted;
      case 'ai_processing': return SubmissionStatus.aiProcessing;
      case 'pending_review': return SubmissionStatus.pendingReview;
      case 'graded': return SubmissionStatus.graded;
      default: return SubmissionStatus.unknown;
    }
  }

  @override
  String toJson(SubmissionStatus status) => switch (status) {
    SubmissionStatus.draft => 'draft',
    SubmissionStatus.submitted => 'submitted',
    SubmissionStatus.aiProcessing => 'ai_processing',
    SubmissionStatus.pendingReview => 'pending_review',
    SubmissionStatus.graded => 'graded',
    SubmissionStatus.unknown => 'unknown',
  };
}
```

Update the `Submission` Freezed class to use the converter:
```dart
@SubmissionStatusConverter()
@JsonKey(name: 'status') @Default(SubmissionStatus.draft) SubmissionStatus status,
```

```dart
// AI Analysis Toggle
Consumer(
  builder: (context, ref, _) {
    final notifier = ref.read(distributeAssignmentNotifierProvider.notifier);
    final aiEnabled = ref.watch(distributeAssignmentNotifierProvider.select((s) => s.aiEnabled));
    final requireReview = ref.watch(distributeAssignmentNotifierProvider.select((s) => s.requireReview));
    
    return Column(
      children: [
        SwitchListTile(
          title: const Text('AI Phân tích bài làm'),
          subtitle: const Text('Sau khi nộp, AI giải thích đáp án và phân tích học lực'),
          secondary: const Icon(Icons.auto_awesome),
          value: aiEnabled,
          onChanged: (_) => notifier.toggleAiEnabled(),
        ),
        AnimatedSize(
          duration: const Duration(milliseconds: 200),
          curve: Curves.easeInOut,
          child: aiEnabled
              ? Container(
                  margin: const EdgeInsets.only(left: 16, right: 16, bottom: 8),
                  padding: const EdgeInsets.all(DesignSpacing.sm),
                  decoration: BoxDecoration(
                    color: DesignColors.surface,
                    borderRadius: BorderRadius.circular(DesignRadius.sm),
                  ),
                  child: Column(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      SwitchListTile(
                        title: const Text('Chờ giáo viên duyệt trước khi công bố'),
                        value: requireReview,
                        onChanged: (_) => notifier.toggleRequireReview(),
                        dense: true,
                      ),
                      Padding(
                        padding: const EdgeInsets.symmetric(
                          horizontal: DesignSpacing.sm,
                          vertical: DesignSpacing.xs,
                        ),
                        child: Text(
                          requireReview
                              ? 'AI phân tích xong → giáo viên xem xét → công bố điểm'
                              : 'AI phân tích xong → tự động công bố điểm',
                          style: DesignTypography.labelSmall.copyWith(
                            color: DesignColors.textSecondary,
                          ),
                        ),
                      ),
                    ],
                  ),
                )
              : const SizedBox.shrink(),
        ),
      ],
    );
  },
),
```
