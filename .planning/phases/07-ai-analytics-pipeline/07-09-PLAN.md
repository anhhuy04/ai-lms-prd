```sql
-- Phase 7: RPC for class average skill mastery (D-22)
-- Returns average mastery per learning objective for all students in a class
-- SECURITY DEFINER to bypass RLS (teacher needs to see all students)

CREATE OR REPLACE FUNCTION get_class_average_skill_mastery(p_class_id uuid)
RETURNS TABLE (
  objective_id uuid,
  objective_name text,
  subject_code text,
  avg_mastery numeric,
  total_students bigint,
  students_below_threshold bigint
)
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
BEGIN
  RETURN QUERY
  SELECT
    ssm.objective_id,
    lo.name AS objective_name,
    lo.subject_code,
    AVG(ssm.mastery_level)::numeric AS avg_mastery,
    COUNT(DISTINCT ssm.student_id) AS total_students,
    COUNT(DISTINCT ssm.student_id) FILTER (WHERE ssm.mastery_level < 0.6) AS students_below_threshold
  FROM student_skill_mastery ssm
  JOIN learning_objectives lo ON lo.id = ssm.objective_id
  WHERE ssm.student_id IN (
    SELECT cs.student_id
    FROM class_students cs
    WHERE cs.class_id = p_class_id
  )
  GROUP BY ssm.objective_id, lo.name, lo.subject_code
  ORDER BY avg_mastery ASC;
END;
$$;
```
